### J11: Copy Summary

```typescript
function formatPlanSummary(draft, selections, evaluation): string {
  return [
    `Plan: ${draft.signalHeadline}`,
    `Positions: ${selections.map(s => `${s.ticker} at ${s.allocationPct}%`).join(', ')}`,
    `Expected: reduces loss by ~$${Math.round(evaluation.downsideReductionCad)}`,
    `Generated by Prism · ${new Date().toLocaleDateString()}`
  ].join('\n')
}

const handleCopy = () => {
  navigator.clipboard.writeText(formatPlanSummary(draft, selections, evaluation))
}
```
